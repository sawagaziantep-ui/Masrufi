function fmt(n) {
  return new Intl.NumberFormat("tr-TR", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(Math.abs(n));
}

function todayISO() {
  return new Date().toISOString().slice(0, 10);
}

function ymOf(dateStr) {
  return dateStr.slice(0, 7);
}

function ymAdd(ym, delta) {
  const [y, m] = ym.split("-").map(Number);
  const d = new Date(y, m - 1 + delta, 1);
  return d.toISOString().slice(0, 7);
}

function computeMonthRemaining(transactions, ym) {
  const monthTx = transactions.filter((t) => ymOf(t.date) === ym);
  const income = monthTx.filter((t) => t.type === "income").reduce((s, t) => s + t.amount, 0);
  const expense = monthTx.filter((t) => t.type === "expense").reduce((s, t) => s + t.amount, 0);
  return income - expense;
}

/* Carries the previous month's remaining balance forward as a new transaction
   dated on the 1st of each subsequent month, so every month's totals start from
   zero and only the leftover amount rolls in as a single labeled entry. */
function applyMonthlyCarryover(transactions) {
  if (transactions.length === 0) return transactions;
  const currentYm = todayISO().slice(0, 7);
  const months = [...new Set(transactions.map((t) => ymOf(t.date)))].sort();
  let lastYm = months[months.length - 1];
  if (lastYm >= currentYm) return transactions;

  let updated = [...transactions];
  let cursor = lastYm;
  while (cursor < currentYm) {
    const nextYm = ymAdd(cursor, 1);
    const alreadyCarried = updated.some((t) => t.category === "carryover" && ymOf(t.date) === nextYm);
    if (!alreadyCarried) {
      const remaining = computeMonthRemaining(updated, cursor);
      if (remaining !== 0) {
        updated.push({
          id: "carryover-" + nextYm,
          type: remaining >= 0 ? "income" : "expense",
          amount: Math.abs(remaining),
          category: "carryover",
          note: remaining >= 0 ? "مدور من الشهر الماضي" : "عجز من الشهر الماضي",
          date: nextYm + "-01",
        });
      }
    }
    cursor = nextYm;
  }
  return updated;
}

function monthLabel(date) {
  return new Intl.DateTimeFormat("ar", { month: "long", year: "numeric" }).format(date);
}

function dayLabel(dateStr) {
  const d = new Date(dateStr + "T00:00:00");
  const today = new Date();
  const yesterday = new Date();
  yesterday.setDate(today.getDate() - 1);
  const sameDay = (a, b) =>
    a.getFullYear() === b.getFullYear() && a.getMonth() === b.getMonth() && a.getDate() === b.getDate();
  if (sameDay(d, today)) return "اليوم";
  if (sameDay(d, yesterday)) return "أمس";
  return new Intl.DateTimeFormat("ar", { day: "numeric", month: "long" }).format(d);
}

function fullDateLabel(dateStr) {
  const d = new Date(dateStr + "T00:00:00");
  return new Intl.DateTimeFormat("ar", { day: "numeric", month: "long", year: "numeric" }).format(d);
}

function monthRangeFromYm(ym) {
  const [y, m] = ym.split("-").map(Number);
  const start = ym + "-01";
  const lastDay = new Date(y, m, 0).getDate();
  const end = ym + "-" + String(lastDay).padStart(2, "0");
  return { start, end };
}

/* ---------- storage hook ---------- */
function useTransactions() {
  const [items, setItems] = useState([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    (async () => {
      try {
        const res = await window.storage.get("wallet-transactions");
        if (res && res.value) setItems(JSON.parse(res.value));
      } catch (e) {
        /* nothing stored yet */
      } finally {
        setLoaded(true);
      }
    })();
  }, []);

  useEffect(() => {
    if (!loaded) return;
    window.storage.set("wallet-transactions", JSON.stringify(items)).catch(() => {});
  }, [items, loaded]);

  return [items, setItems, loaded];
}

/* ---------- small UI pieces ---------- */
function Ring({ percent, size = 220 }) {
  const stroke = 16;
  const r = (size - stroke) / 2;
  const c = 2 * Math.PI * r;
  const clamped = Math.max(0, Math.min(100, percent));
  const dash = (clamped / 100) * c;
  return (
    <svg width={size} height={size} style={{ transform: "rotate(-90deg)" }}>
      <circle cx={size / 2} cy={size / 2} r={r} stroke="#E5B842" strokeWidth={stroke} fill="none" opacity={0.9} />
      <circle
        cx={size / 2}
        cy={size / 2}
        r={r}
        stroke="#10B981"
        strokeWidth={stroke}
        fill="none"
        strokeDasharray={`${dash} ${c - dash}`}
        strokeLinecap="round"
      />
    </svg>
  );
}

function Donut({ segments, size = 200, centerLabel, centerValue }) {
  const stroke = 34;
  const r = (size - stroke) / 2;
  const c = 2 * Math.PI * r;
  let offset = 0;
  const total = segments.reduce((s, x) => s + x.value, 0) || 1;
  return (
    <svg width={size} height={size} style={{ transform: "rotate(-90deg)" }}>
      <circle cx={size / 2} cy={size / 2} r={r} stroke="rgba(255,255,255,0.08)" strokeWidth={stroke} fill="none" />
      {segments.map((seg, i) => {
        const len = (seg.value / total) * c;
        const el = (
          <circle
            key={i}
            cx={size / 2}
            cy={size / 2}
            r={r}
            stroke={seg.color}
            strokeWidth={stroke}
            fill="none"
            strokeDasharray={`${len} ${c - len}`}
            strokeDashoffset={-offset}
          />
        );
        offset += len;
        return el;
      })}
      <text
        x="50%"
        y="45%"
        textAnchor="middle"
        fill="#94A3B8"
        fontSize="12"
        fontFamily="Tajawal"
        transform={`rotate(90 ${size / 2} ${size / 2})`}
      >
        {centerLabel}
      </text>
      <text
        x="50%"
        y="58%"
        textAnchor="middle"
        fill="#FFFFFF"
        fontSize="22"
        fontWeight="900"
        fontFamily="Tajawal"
        transform={`rotate(90 ${size / 2} ${size / 2})`}
      >
        {centerValue}
      </text>
    </svg>
  );
}

function AddModal({ onClose, onSave }) {

  const [category, setCategory] = useState(EXPENSE_CATEGORIES[2].id);
  const [note, setNote] = useState("");

    setCategory(type === "expense" ? EXPENSE_CATEGORIES[2].id : INCOME_CATEGORIES[0].id);
  }, [type]);

  const cats = type === "expense" ? EXPENSE_CATEGORIES : INCOME_CATEGORIES;

  function press(key) {
    if (key === "back") {
      setDisplay((d) => (d.length > 1 ? d.slice(0, -1) : "0"));
      return;
    }
    if (key === ".") {
      if (display.includes(".")) return;
      setDisplay((d) => d + ".");
      return;
    }
    setDisplay((d) => {
      if (d === "0") return key;
      if (d.includes(".") && d.split(".")[1].length >= 2) return d;
      return d + key;

    const amount = parseFloat(display);

      id: Date.now().toString(),
      type,
      amount,
      category,
      note: note.trim(),
      date: todayISO(),

  const filtered = transactions.filter((t) => filter === "all" || t.type === (filter === "income" ? "income" : "expense"));

  const groups = useMemo(() => {

    [...filtered]
      .sort((a, b) => (a.date < b.date ? 1 : -1))
      .forEach((t) => {
        if (!map[t.date]) map[t.date] = [];
        map[t.date].push(t);
      });
    return Object.entries(map);
  }, [filtered]);

  return (
    <div style={{ padding: "24px 18px 100px" }}>
      <div style={{ fontSize: 26, fontWeight: 900, color: "#FFFFFF", textAlign: "center", marginBottom: 16 }}>
        الحركات
      </div>
      <div style={{ display: "flex", gap: 8, justifyContent: "center", marginBottom: 16 }}>
        {[
          { id: "all", label: "الكل" },
          { id: "expense", label: "مصاريف" },
          { id: "income", label: "دخل" },
        ].map((f) => (
          <button
            key={f.id}
            onClick={() => setFilter(f.id)}
            style={{
              ...pillStyle,
              background: filter === f.id ? "#E5B842" : "#1E293B",
              color: filter === f.id ? "#101728" : "#94A3B8",
            }}
          >
            {f.label}
          </button>
        ))}
      </div>

      <div style={{ color: total >= 0 ? "#10B981" : "#EF4444", fontWeight: 900, fontSize: 16, marginBottom: 8 }}>
        {total >= 0 ? "+" : "-"}
        {fmt(total)} {CURRENCY}
      </div>

      {groups.length === 0 && (
        <div style={{ textAlign: "center", color: "#64748B", fontSize: 13, padding: "40px 0" }}>
          ما في حركات مسجّلة
        </div>
      )}

      {groups.map(([date, txs]) => (
        <div key={date} style={{ marginBottom: 14 }}>
          <div style={{ fontSize: 13, color: "#94A3B8", marginBottom: 4 }}>{dayLabel(date)}</div>
          <div style={{ ...cardStyle, padding: "4px 14px" }}>
            {txs.map((t) => (
              <TxRow key={t.id} tx={t} onDelete={onDelete} />
            ))}
          </div>
        </div>
      ))}
    </div>
  );
}

function PrintReport({ transactions, range, onFinished }) {
  useEffect(() => {
    if (!range) return;
    const t = setTimeout(() => window.print(), 250);
    const handler = () => onFinished();
    window.addEventListener("afterprint", handler);
    return () => {
      clearTimeout(t);
      window.removeEventListener("afterprint", handler);
    };
  }, [range]);

  if (!range) return null;

  const rangeTx = transactions
    .filter((t) => t.date >= range.start && t.date <= range.end)
    .sort((a, b) => (a.date < b.date ? -1 : 1));

  const income = rangeTx.filter((t) => t.type === "income").reduce((s, t) => s + t.amount, 0);
  const expense = rangeTx.filter((t) => t.type === "expense").reduce((s, t) => s + t.amount, 0);

  const byCat = {};
  rangeTx.filter((t) => t.type === "expense").forEach((t) => (byCat[t.category] = (byCat[t.category] || 0) + t.amount));
  const catRows = Object.entries(byCat).map(([id, value]) => ({ ...catById(id), value })).sort((a, b) => b.value - a.value);

  return (
    <div id="print-report-root">
      <div className="pr-page">
        <div className="pr-header">
          <div>
            <div className="pr-title">تقرير المصروفات</div>
            <div className="pr-sub">مصاريفي</div>
          </div>
          <div className="pr-period">
            <div style={{ fontWeight: 700, marginBottom: 4 }}>فترة التقرير</div>
            <div className="pr-period-row"><span>تاريخ البداية</span><b>{fullDateLabel(range.start)}</b></div>
            <div className="pr-period-row"><span>تاريخ النهاية</span><b>{fullDateLabel(range.end)}</b></div>
          </div>
        </div>

        <table className="pr-table">
          <thead>
            <tr>
              <th>التاريخ</th>
              <th>الوصف</th>
              <th>التصنيف</th>
              <th>المبلغ</th>
            </tr>
          </thead>
          <tbody>
            {rangeTx.length === 0 && (
              <tr><td colSpan="4" style={{ textAlign: "center", padding: 16, color: "#94A3B8" }}>ما في حركات بهاي الفترة</td></tr>
            )}
            {rangeTx.map((t) => (
              <tr key={t.id}>
                <td>{fullDateLabel(t.date)}</td>
                <td>{t.note || catById(t.category).label}</td>
                <td>{catById(t.category).label}</td>
                <td className={t.type === "income" ? "pr-green" : "pr-red"}>
                  {t.type === "income" ? "+" : "-"}{fmt(t.amount)} {CURRENCY}
                </td>
              </tr>
            ))}
          </tbody>
        </table>

        <div className="pr-bar">إجمالي الفئات</div>
        <table className="pr-table">
          <tbody>
            {catRows.length === 0 && (
              <tr><td colSpan="2" style={{ textAlign: "center", padding: 12, color: "#94A3B8" }}>لا يوجد مصاريف</td></tr>
            )}
            {catRows.map((c) => (
              <tr key={c.id}>
                <td>{c.label}</td>
                <td className="pr-red">{fmt(c.value)} {CURRENCY}</td>
              </tr>
            ))}
          </tbody>
        </table>

        <div className="pr-totals">
          <div className="pr-total-row"><span>إجمالي الدخل</span><b className="pr-green">{fmt(income)} {CURRENCY}</b></div>
          <div className="pr-total-row"><span>إجمالي المصاريف</span><b className="pr-red">{fmt(expense)} {CURRENCY}</b></div>
          <div className="pr-total-row pr-grand"><span>الصافي</span><b>{fmt(income - expense)} {CURRENCY}</b></div>
        </div>

        <div className="pr-footer">تم إنشاء هذا التقرير من تطبيق مصاريفي</div>
      </div>
    </div>
  );
}

function ReportsScreen({ transactions }) {
  const [monthOffset, setMonthOffset] = useState(0);
  const [exportOpen, setExportOpen] = useState(false);
  const [exportMode, setExportMode] = useState("lastMonth");
  const [customStart, setCustomStart] = useState("");
  const [customEnd, setCustomEnd] = useState("");
  const [printRange, setPrintRange] = useState(null);

  function handleExport() {
    let range;
    if (exportMode === "custom") {
      if (!customStart || !customEnd) return;
      range = { start: customStart, end: customEnd };
    } else {
      const now = new Date();
      now.setDate(1);
      now.setMonth(now.getMonth() - 1);
      range = monthRangeFromYm(now.toISOString().slice(0, 7));
    }
    setPrintRange(range);
    setExportOpen(false);
  }

  const base = new Date();
  base.setDate(1);
  base.setMonth(base.getMonth() + monthOffset);
  const key = base.toISOString().slice(0, 7);

  const monthTx = transactions.filter((t) => t.date.slice(0, 7) === key);
  const income = monthTx.filter((t) => t.type === "income").reduce((s, t) => s + t.amount, 0);
  const expense = monthTx.filter((t) => t.type === "expense").reduce((s, t) => s + t.amount, 0);
  const remaining = income - expense;
  const savingsPct = income > 0 ? Math.max(0, Math.min(100, (remaining / income) * 100)) : 0;

  const byCat = useMemo(() => {
    const map = {};
    monthTx
      .filter((t) => t.type === "expense")
      .forEach((t) => (map[t.category] = (map[t.category] || 0) + t.amount));
    return Object.entries(map)
      .map(([id, value]) => ({ ...catById(id), value }))
      .sort((a, b) => b.value - a.value);
  }, [monthTx]);

  return (
    <>
    <div style={{ padding: "24px 18px 100px" }}>
      <div style={{ fontSize: 26, fontWeight: 900, color: "#FFFFFF", textAlign: "center", marginBottom: 16 }}>
        التقارير
      </div>

      <button
        onClick={() => setExportOpen((o) => !o)}
        style={{ ...pillStyle, display: "flex", alignItems: "center", gap: 8, background: "#1E293B", color: "#E5B842", margin: "0 auto 14px", padding: "10px 18px" }}
      >
        <Download size={16} />
        تحميل تقرير PDF
      </button>

      {exportOpen && (
        <div style={{ ...cardStyle, marginBottom: 16 }}>
          <div style={{ display: "flex", gap: 8, marginBottom: 12 }}>
            <button
              onClick={() => setExportMode("lastMonth")}
              style={{ ...pillStyle, flex: 1, background: exportMode === "lastMonth" ? "#E5B842" : "#182032", color: exportMode === "lastMonth" ? "#101728" : "#94A3B8" }}
            >
              الشهر الماضي
            </button>
            <button
              onClick={() => setExportMode("custom")}
              style={{ ...pillStyle, flex: 1, background: exportMode === "custom" ? "#E5B842" : "#182032", color: exportMode === "custom" ? "#101728" : "#94A3B8" }}
            >
              فترة مخصصة
            </button>
          </div>

          {exportMode === "custom" && (
            <div style={{ display: "flex", gap: 8, marginBottom: 12 }}>
              <input
                type="date"
                value={customStart}
                onChange={(e) => setCustomStart(e.target.value)}
                style={{ flex: 1, background: "#101728", border: "1px solid rgba(255,255,255,0.08)", borderRadius: 8, padding: 10, color: "#FFFFFF", fontFamily: "Tajawal" }}
              />
              <input
                type="date"
                value={customEnd}
                onChange={(e) => setCustomEnd(e.target.value)}
                style={{ flex: 1, background: "#101728", border: "1px solid rgba(255,255,255,0.08)", borderRadius: 8, padding: 10, color: "#FFFFFF", fontFamily: "Tajawal" }}
              />
            </div>
          )}

          <button
            onClick={handleExport}
            style={{ width: "100%", background: "#E5B842", border: "none", borderRadius: 10, padding: "12px 0", fontWeight: 900, color: "#101728", cursor: "pointer" }}
          >
            تصدير PDF
          </button>
        </div>
      )}

      <div style={{ ...cardStyle, display: "flex", alignItems: "center", justifyContent: "space-between", marginBottom: 16 }}>
        <button onClick={() => setMonthOffset((m) => m - 1)} style={iconBtnStyle}>
          <ChevronRight size={18} color="#FFFFFF" />
        </button>
        <div style={{ fontWeight: 900, fontSize: 17, color: "#FFFFFF" }}>{monthLabel(base)}</div>
        <button
          onClick={() => setMonthOffset((m) => Math.min(0, m + 1))}
          style={iconBtnStyle}
          disabled={monthOffset === 0}
        >
          <ChevronLeft size={18} color={monthOffset === 0 ? "#3A4A70" : "#FFFFFF"} />
        </button>
      </div>

      <div style={{ display: "flex", gap: 10, marginBottom: 16 }}>
        <div style={{ ...cardStyle, textAlign: "center" }}>
          <Wallet size={18} color="#94A3B8" style={{ marginBottom: 6 }} />
          <div style={{ fontSize: 12, color: "#94A3B8" }}>المتبقي</div>
          <div style={{ fontSize: 16, fontWeight: 900, color: "#FFFFFF" }}>{fmt(remaining)}</div>
        </div>
        <div style={{ ...cardStyle, textAlign: "center" }}>
          <ArrowDown size={18} color="#EF4444" style={{ marginBottom: 6 }} />
          <div style={{ fontSize: 12, color: "#94A3B8" }}>المصاريف</div>
          <div style={{ fontSize: 16, fontWeight: 900, color: "#FFFFFF" }}>{fmt(expense)}</div>
        </div>
        <div style={{ ...cardStyle, textAlign: "center" }}>
          <ArrowUp size={18} color="#10B981" style={{ marginBottom: 6 }} />
          <div style={{ fontSize: 12, color: "#94A3B8" }}>الدخل</div>
          <div style={{ fontSize: 16, fontWeight: 900, color: "#FFFFFF" }}>{fmt(income)}</div>
        </div>
      </div>

      <div style={{ ...cardStyle, marginBottom: 16 }}>
        <div style={{ display: "flex", justifyContent: "space-between", marginBottom: 8 }}>
          <span style={{ fontWeight: 900, color: "#10B981" }}>{savingsPct.toFixed(0)}%</span>
          <span style={{ fontSize: 13, color: "#94A3B8" }}>نسبة الادخار</span>
        </div>
        <div style={{ height: 8, borderRadius: 999, background: "rgba(255,255,255,0.08)" }}>
          <div style={{ height: 8, borderRadius: 999, width: `${savingsPct}%`, background: "#10B981" }} />
        </div>
      </div>

      <div style={{ fontSize: 15, fontWeight: 900, color: "#FFFFFF", marginBottom: 10 }}>توزيع المصاريف</div>
      <div style={cardStyle}>
        {byCat.length === 0 ? (
          <div style={{ textAlign: "center", color: "#64748B", fontSize: 13, padding: "20px 0" }}>
            ما في مصاريف هالشهر
          </div>
        ) : (
          <>
            <div style={{ display: "flex", justifyContent: "center", marginBottom: 14 }}>
              <Donut segments={byCat} centerLabel="الإجمالي" centerValue={fmt(expense)} />
            </div>
            {byCat.map((c) => {
              const pct = expense > 0 ? (c.value / expense) * 100 : 0;
              return (
                <div key={c.id} style={{ marginBottom: 10 }}>
                  <div style={{ display: "flex", justifyContent: "space-between", fontSize: 13, marginBottom: 4 }}>
                    <span style={{ display: "flex", alignItems: "center", gap: 8, color: "#FFFFFF" }}>
                      <span style={{ width: 10, height: 10, borderRadius: 999, background: c.color, display: "inline-block" }} />
                      {c.label}
                    </span>
                    <span style={{ color: "#94A3B8" }}>
                      {fmt(c.value)} · {pct.toFixed(0)}%
                    </span>
                  </div>
                  <div style={{ height: 6, borderRadius: 999, background: "rgba(255,255,255,0.08)" }}>
                    <div style={{ height: 6, borderRadius: 999, width: `${pct}%`, background: c.color }} />
                  </div>
                </div>
              );
            })}
          </>
        )}
      </div>
    </div>

    <PrintReport transactions={transactions} range={printRange} onFinished={() => setPrintRange(null)} />
    </>
  );
}
